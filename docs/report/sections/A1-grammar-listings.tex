\section{Grammar Reference}\label{app:grammar}

\subsection{Terminal classes}

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{Terminal} & \textbf{Category} & \textbf{Notes} \\
\midrule
\term{DET}     & Determiner    & Triggers left factoring in \nonterm{NP} \\
\term{NOUN}    & Noun          & Largest open class after \term{VERB} \\
\term{PRON}    & Pronoun       & Full pronouns only; clitics not yet split out \\
\term{VERB}    & Verb          & Chains via \nonterm{VerbTail} for serial verbs \\
\term{PREP}    & Preposition   & Heads \nonterm{PP}; allows clause fronting \\
\term{ADJ}     & Adjective     & Pre-nominal in \nonterm{NP$'$}, or a complement \\
\term{ADV}     & Adverb        & Pre-verbal or inside the verb group \\
\term{NEG}     & Negation      & Split from \term{ADV}: positionally constrained \\
\term{CONJ}    & Conjunction   & Joins units in \nonterm{Seq} \\
\term{SEP}     & Separator     & \term{,} \term{;} \term{:} --- clause boundaries \\
\term{INTJ}    & Interjection  & May stand alone as a complete utterance \\
\term{FORMULA} & Fixed expression & Frozen multiword unit, lexically matched \\
\bottomrule
\end{tabular}
\end{center}

\subsection{Nonterminals}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Nonterminal} & \textbf{Role} \\
\midrule
\nonterm{S}          & Start symbol \\
\nonterm{Seq}        & Sequence of units joined by conjunctions or separators \\
\nonterm{Unit}       & One utterance: interjection, formula, or clause \\
\nonterm{Clause}     & Fronted \nonterm{PP}, subject clause, or bare predicate \\
\nonterm{ClauseOpt}  & Optional clause after a fronted \nonterm{PP} \\
\nonterm{ClauseTail} & What may follow a subject \nonterm{NP} \\
\nonterm{Predicate}  & Adverbs, optional negation, verb group and complements \\
\nonterm{NegOpt}     & Optional pre-verbal negation \\
\nonterm{VerbGroup}  & Head verb plus its tail \\
\nonterm{VerbTail}   & Serial verbs, interleaved adverbs and negations \\
\nonterm{Comps}      & Complement list after the verb group \\
\nonterm{CompObj}    & Object of a preposition: \nonterm{NP} or \nonterm{VerbGroup} \\
\nonterm{NP}         & Noun phrase \\
\nonterm{PP}         & Prepositional phrase \\
\midrule
\nonterm{Seq$'$}, \nonterm{Seq$''$} & Generated by transformation \\
\nonterm{Unit$'$}, \nonterm{NP$'$}  & Generated by transformation \\
\bottomrule
\end{tabular}
\end{center}

\subsection{Reproducing every table in this report}

All data tables are exported from the live analyzer. To regenerate them:

\begin{lstlisting}[language=bash, numbers=none]
cd docs/report
python tools/export_tables.py     # regenerate generated/*.tex
python tools/render_diagrams.py   # regenerate figures/*.png from diagrams/*.puml
latexmk -pdf main.tex             # build the report
\end{lstlisting}

or run \texttt{pwsh ./build.ps1}, which performs all three steps in order.
